cmake_minimum_required(VERSION 3.20)
project(SunrisePortableTests LANGUAGES CXX)
enable_testing()
add_executable(activity_host_selection_test activity_host_selection_test.cpp)
target_compile_features(activity_host_selection_test PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(activity_host_selection_test PRIVATE /W4 /WX /UNDEBUG)
else()
    target_compile_options(activity_host_selection_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME activity_host_selection COMMAND activity_host_selection_test)
add_executable(socket_row_relocation_test socket_row_relocation_test.cpp)
target_compile_features(socket_row_relocation_test PRIVATE cxx_std_20)

else()
    target_compile_options(socket_row_relocation_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME socket_row_relocation
    COMMAND socket_row_relocation_test "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/socket_rows.hex")
